class BoneDataset(Dataset):
    
    def __init__(self,df,img_dir,transform=None):
        
        self.df=df
        self.img_dir=img_dir
        self.transform=transform
        
    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        try:
            row = self.df.iloc[idx]
            img_name = row['filename']
            label = row['target_encoded']
            
            img_path = os.path.join(self.img_dir, img_name)           
            image = Image.open(img_path).convert('RGB')
            
            if self.transform:
                image = self.transform(image)
                
            return image, torch.tensor(label, dtype=torch.long)
            
        except Exception as e:
            print(f"UYARI: Görüntü okunamadı (idx={idx}, {img_name}) -> Hata: {e}")
            dummy_image = Image.new('RGB', (300, 300), (0, 0, 0))
            if self.transform:
                dummy_image = self.transform(dummy_image)
            return dummy_image, torch.tensor(label, dtype=torch.long)


def get_dataLoaders(batch_size,train_df,test_df,norm_mean,norm_std,img_dir):
    
    transform_train=transforms.Compose([

    ])
    
    transform_test=transforms.Compose([

    ])
    
    train_ds=BoneDataset(df=train_df,img_dir=img_dir,transform=transform_train)

                            batch_size=batch_size,
                            shuffle=True,

                            pin_memory=True)
    
    test_loader=DataLoader(dataset=test_ds,
                           batch_size=batch_size,
                           shuffle=False,
                           num_workers=2,
                           pin_memory=True)
    
    return train_loader,test_loader

def load_model(device, num_classes):
    
    weight = models.EfficientNet_B3_Weights.IMAGENET1K_V1
    model = models.efficientnet_b3(weights=weight)
    
    # Başlangıçta tüm parametreleri dondur (Transfer Learning için)
    for param in model.parameters():
        param.requires_grad = False
    
    # Son 3-4 bloku açmak için (Fine-tuning hazırlığı)
    block_unfreeze = [6, 7, 8]
    
    in_features = model.classifier[1].in_features
    
    # Yeni classifier (başlık) - daha iyi regularizasyon
    model.classifier = nn.Sequential(
        nn.Linear(in_features, 512),
        nn.BatchNorm1d(512),
        nn.ReLU(),
        nn.Dropout(0.5),
        nn.Linear(512, 256),
        nn.BatchNorm1d(256),
        nn.ReLU(),
        nn.Dropout(0.3),
        nn.Linear(256, num_classes)
    )
    
    model = model.to(device)
    
    return model, block_unfreeze


def train_model(model, train_loader, valid_loader, criterion, optimizer, num_epochs, device, scheduler=None, stage_name=""):
    
    print(f"\n{'='*60}")
    print(f"{stage_name} - Egitim basladi")
    print(f"{'='*60}")
    
    model.train()
    
    train_losses = []
    train_accuracies = []
    valid_losses = []
    valid_accuracies = []
    
    scaler = GradScaler(enabled=True) 
    
    best_loss = float('inf')
    
    for epoch in range(num_epochs):
        
        # ===== TRAIN LOOP =====
        total_loss = 0
        correct = 0
        total_samples = 0
        
        loop = tqdm(train_loader, desc=f"[{stage_name}] Epoch {epoch+1}/{num_epochs}")
        
        for images, labels in loop:
            images, labels = images.to(device), labels.to(device).long()
            
            optimizer.zero_grad()
            
            with autocast(device_type='cuda', enabled=True):
                predictions = model(images)
                loss = criterion(predictions, labels)
            
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            
            total_loss += loss.item() * images.size(0)
            _, predicted = torch.max(predictions.data, 1)
            correct += (predicted == labels).sum().item()
            total_samples += labels.size(0)
            
            loop.set_postfix(loss=loss.item())
        
        train_avg_loss = total_loss / total_samples
        train_accuracy = (correct / total_samples) * 100
        
        train_losses.append(train_avg_loss)
        train_accuracies.append(train_accuracy)
        
        # ===== VALIDATION LOOP =====
        model.eval()
        valid_total_loss = 0
        valid_correct = 0
        valid_total_samples = 0
        
        with torch.no_grad():
            for images, labels in valid_loader:
                images, labels = images.to(device), labels.to(device).long()
                
                predictions = model(images)
                loss = criterion(predictions, labels)
                
                valid_total_loss += loss.item() * images.size(0)
                _, predicted = torch.max(predictions.data, 1)
                valid_correct += (predicted == labels).sum().item()
                valid_total_samples += labels.size(0)
        
        valid_avg_loss = valid_total_loss / valid_total_samples
        valid_accuracy = (valid_correct / valid_total_samples) * 100
        
        valid_losses.append(valid_avg_loss)
        valid_accuracies.append(valid_accuracy)
        
        model.train()
        
        # ===== MODEL KAYDETME =====
        if valid_avg_loss < best_loss:
            best_loss = valid_avg_loss
            torch.save(model.state_dict(), f"best_model_{stage_name}.pth")
            print(f"--> En iyi model kaydedildi! (Valid Loss: {best_loss:.4f})")
        
        # ===== SCHEDULER =====
        if scheduler is not None:
            if isinstance(scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
                scheduler.step(valid_avg_loss)
            else:
                scheduler.step()
        
        current_lr = optimizer.param_groups[0]['lr']
        print(f"Epoch {epoch+1}/{num_epochs} | Train Loss: {train_avg_loss:.4f} | Train Acc: {train_accuracy:.2f}% | Valid Loss: {valid_avg_loss:.4f} | Valid Acc: {valid_accuracy:.2f}% | LR: {current_lr:.6f}")
    
    torch.save(model.state_dict(), f"last_model_{stage_name}.pth")
    print(f"\n{stage_name} egitimi tamamlandi. 'last_model_{stage_name}.pth' kaydedildi.")
    
    return train_losses, train_accuracies, valid_losses, valid_accuracies


def unfreeze_blocks(model, block_indices):
    """
    Belirtilen blokların kilidini aç (backprop için)
    """
    for i, child in enumerate(model.features.children()):
        if i in block_indices:
            for param in child.parameters():
                param.requires_grad = True
            print(f"Block {i} kilidinin açıldı.")


def calculate_class_weights(df):
    """
    Eğitim setinde sınıf dengesizliğini gidermek için class weights hesapla
    
    Args:
        df: target_encoded sütunu olan DataFrame
    
    Returns:
        torch.Tensor: Normalize edilmiş class weights
    """
    # Sınıf frekanslarını hesapla
    class_counts = df['target_encoded'].value_counts().sort_index()
    total_samples = len(df)
    
    # Her sınıf için ağırlık (inverse frequency)
    class_weights = total_samples / (len(class_counts) * class_counts.values)
    class_weights = torch.tensor(class_weights, dtype=torch.float32)
    
    print("\n--- Sınıf Ağırlıkları (Class Weights) ---")
    for idx, weight in enumerate(class_weights):
        print(f"Sınıf {idx}: {weight:.4f} (Örnek sayısı: {class_counts[idx]})")
    
    return class_weights

def test_model(model, test_loader, device, dataset_type, class_name):
    """
    Test Time Augmentation (TTA) ile model test etme
    Her görüntüyü 5 farklı varyasyonda işleyip ortalamasını al
    """
    
    model.eval()
    test_correct = 0
    test_total = 0
    all_predictions = []
    all_labels = []
    
    # Normalize için sabitler
    NORM_MEAN = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1).to(device)
    NORM_STD = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1).to(device)
    
    # PIL'den Tensor'a dönüşüm (TTA için)
    to_tensor = transforms.ToTensor()
    
    def apply_normalize(tensor_img):
        """Tensor üzerine normalize uygula"""
        return (tensor_img - NORM_MEAN.cpu()) / NORM_STD.cpu()
    
    def denormalize(tensor_img):
        """Tensor'ı denormalize et (PIL'e çevirmek için)"""
        tensor_img = tensor_img.to(device)
        return torch.clamp((tensor_img * NORM_STD.cpu() + NORM_MEAN.cpu()), 0, 1)
    
    to_pil = transforms.ToPILImage()
    
    with torch.no_grad():
        for images_tensor, labels in tqdm(test_loader, desc=f"Testing {dataset_type} (TTA)"):
            images_tensor = images_tensor.to(device)
            labels = labels.to(device).long()
            
            batch_probs = []
            
            for img_idx in range(images_tensor.shape[0]):
                try:
                    # Normalize edilmiş tensorü denormalize et
                    img_denorm = denormalize(images_tensor[img_idx].cpu()).clamp(0, 1)
                    img_pil = to_pil(img_denorm)
                    
                    # 5 farklı TTA varyasyonu
                    tta_outputs = []
                    
                    # 1. Orijinal
                    augmented = to_tensor(img_pil).to(device)
                    augmented = apply_normalize(augmented).unsqueeze(0)
                    with autocast(device_type='cuda' if device.type == 'cuda' else 'cpu'):
                        output = model(augmented)
                    tta_outputs.append(torch.softmax(output, dim=1).cpu())
                    
                    # 2. Yatay çevirme
                    augmented = to_tensor(F.hflip(img_pil)).to(device)
                    augmented = apply_normalize(augmented).unsqueeze(0)
                    with autocast(device_type='cuda' if device.type == 'cuda' else 'cpu'):
                        output = model(augmented)
                    tta_outputs.append(torch.softmax(output, dim=1).cpu())
                    
                    # 3. Dikey çevirme
                    augmented = to_tensor(F.vflip(img_pil)).to(device)
                    augmented = apply_normalize(augmented).unsqueeze(0)
                    with autocast(device_type='cuda' if device.type == 'cuda' else 'cpu'):
                        output = model(augmented)
                    tta_outputs.append(torch.softmax(output, dim=1).cpu())
                    
                    # 4. +5 Derece rotasyon
                    augmented = to_tensor(F.rotate(img_pil, 5)).to(device)
                    augmented = apply_normalize(augmented).unsqueeze(0)
                    with autocast(device_type='cuda' if device.type == 'cuda' else 'cpu'):
                        output = model(augmented)
                    tta_outputs.append(torch.softmax(output, dim=1).cpu())
                    
                    # 5. -5 Derece rotasyon
                    augmented = to_tensor(F.rotate(img_pil, -5)).to(device)
                    augmented = apply_normalize(augmented).unsqueeze(0)
                    with autocast(device_type='cuda' if device.type == 'cuda' else 'cpu'):
                        output = model(augmented)
                    tta_outputs.append(torch.softmax(output, dim=1).cpu())
                    
                    # Varyasyonların ortalamasını al
                    avg_prob = torch.mean(torch.stack(tta_outputs), dim=0)
                    batch_probs.append(avg_prob)
                
                except Exception as e:
                    print(f"  ⚠ TTA hatası (idx={img_idx}): {e} - Normal tahmine fallback")
                    # Fallback: Direkt tahmin
                    with torch.no_grad():
                        output = model(images_tensor[img_idx:img_idx+1])
                    batch_probs.append(torch.softmax(output, dim=1).cpu())
            
            batch_probs = torch.cat(batch_probs, dim=0)
            _, predicted = torch.max(batch_probs, 1)
            
            test_total += labels.size(0)
            test_correct += (predicted == labels.cpu()).sum().item()
            all_predictions.extend(predicted.numpy())
            all_labels.extend(labels.cpu().numpy())
    
    test_accuracy = 100 * test_correct / test_total
    print(f"\n{dataset_type} Doğruluğu (TTA ile) = {test_accuracy:.2f}%")
    
    print("\nClassification Report:")
    print(classification_report(all_labels, all_predictions, target_names=class_name, zero_division=0))
    
    cm = confusion_matrix(all_labels, all_predictions)
    
    plt.figure(figsize=(10, 8))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
                xticklabels=class_name, yticklabels=class_name, cbar_kws={'label': 'Sayı'})
    
    plt.title(f'Konfüzyon Matrisi ({dataset_type}) - TTA ile')
    plt.ylabel('Gerçek Sınıf (True Label)')
    plt.xlabel('Tahmin Edilen Sınıf (Predicted Label)')
    plt.tight_layout()
    plt.show()
    
    return test_accuracy

    test_df=pd.read_csv(os.path.join(base_dir, "test.csv"))
    valid_df=pd.read_csv(os.path.join(base_dir, "valid.csv"))
    
    # target_encoded sütununu oluştur (string -> integer)
    unique_classes = sorted(train_df['target'].unique())
    class_to_idx = {cls: idx for idx, cls in enumerate(unique_classes)}
    idx_to_class = {idx: cls for cls, idx in class_to_idx.items()}
    
    train_df['target_encoded'] = train_df['target'].map(class_to_idx)
    test_df['target_encoded'] = test_df['target'].map(class_to_idx)
    valid_df['target_encoded'] = valid_df['target'].map(class_to_idx)
    
    num_classes=len(unique_classes)
    class_name=unique_classes
    
    print("Siniflarin haritasi= ",idx_to_class)
    
    print("\n-----Train Test Valid Bolunme Bilgileri------")
    print("Train Veri Sayisi=",len(train_df))
    print("Test Veri Sayisi=",len(test_df))
    print("Valid Veri Sayisi=",len(valid_df))
    
    # ===== SINIF AĞIRLIKLARI HESAPLA =====
    class_weights = calculate_class_weights(train_df)
    class_weights = class_weights.to(device)
    
    
    norm_mean = [0.485, 0.456, 0.406]
    norm_std = [0.229, 0.224, 0.225]
    
    batch_size=16
    
    # Train loader
    train_img_dir=os.path.join(base_dir, "train")
    train_loader, _ = get_dataLoaders(batch_size=batch_size,
                                      train_df=train_df,
                                      test_df=train_df,  # Dummy, kullanılmayacak
                                      norm_mean=norm_mean,
                                      norm_std=norm_std,
                                      img_dir=train_img_dir)
    
    # Test loader
    test_img_dir=os.path.join(base_dir, "test")
    _, test_loader = get_dataLoaders(batch_size=batch_size,
                                     train_df=test_df,  # Dummy, kullanılmayacak
                                     test_df=test_df,
                                     norm_mean=norm_mean,
                                     norm_std=norm_std,
                                     img_dir=test_img_dir)
    
    # Valid loader
    valid_img_dir=os.path.join(base_dir, "valid")
    _, valid_loader = get_dataLoaders(batch_size=batch_size,
                                      train_df=valid_df,  # Dummy, kullanılmayacak
                                      test_df=valid_df,
                                      norm_mean=norm_mean,
                                      norm_std=norm_std,
                                      img_dir=valid_img_dir)
    
    dataiter=iter(train_loader)
    images,labels=next(dataiter)
    n=10
    
    visualization(n,images,labels,class_name)
    
    # ===== MODEL YÜKLEMESİ =====
    model, block_unfreeze = load_model(device, num_classes)
    
    print("\n" + "="*60)
    print("AŞAMA 1: PREDİK BAŞLIĞINI EĞİT (HEAD ONLY)")
    print("="*60)
    print("Modelin gövdesi donmuş, sadece yeni classifier eğitilecek")
    print("Epoch: 5-10, Learning Rate: 5e-4")
    
    # Loss fonksiyonu - Label Smoothing + Class Weights ile
    criterion = nn.CrossEntropyLoss(label_smoothing=0.1, weight=class_weights)
    
    # Sadece classifier parametrelerini optimize et (Aşama 1)
    optimizer_stage1 = optim.AdamW(model.classifier.parameters(), lr=5e-4, weight_decay=0.001)
    scheduler_stage1 = optim.lr_scheduler.ReduceLROnPlateau(
        optimizer_stage1,
        mode='min',
        factor=0.1,
        patience=7,
        verbose=True,
        min_lr=1e-6
    )
    
    # Aşama 1 eğitimi
    train_losses_s1, train_accs_s1, valid_losses_s1, valid_accs_s1 = train_model(
        model, train_loader, valid_loader,
        criterion, optimizer_stage1, 
        num_epochs=10,
        device=device,
        scheduler=scheduler_stage1,
        stage_name="STAGE1_HEAD"
    )
    
    print("\n" + "="*60)
    print("AŞAMA 2: FINE-TUNING (SON BLOKLAR + YENİ BAŞLIK)")
    print("="*60)
    print("Modelin son 3-4 bloğunun kilidini aç")
    print("Epoch: 30-50, Learning Rate: 1e-5")
    
    # Son blokları aç
    unfreeze_blocks(model, block_unfreeze)
    print(f"Unfreeze bloklar: {block_unfreeze}")
    
    # Tüm model parametrelerini optimize et (Aşama 2)
    optimizer_stage2 = optim.AdamW(
        [
            {'params': model.features.parameters(), 'lr': 1e-5},  # SON BLOKLAR
            {'params': model.classifier.parameters(), 'lr': 1e-4}  # YENİ BAŞLIK
        ],
        weight_decay=0.001
    )
    
    scheduler_stage2 = optim.lr_scheduler.ReduceLROnPlateau(
        optimizer_stage2,
        mode='min',
        factor=0.1,
        patience=7,
        verbose=True,
        min_lr=1e-7
    )
    
    # Aşama 2 eğitimi
    train_losses_s2, train_accs_s2, valid_losses_s2, valid_accs_s2 = train_model(
        model, train_loader, valid_loader,
        criterion, optimizer_stage2,
        num_epochs=50,
        device=device,
        scheduler=scheduler_stage2,
        stage_name="STAGE2_FINETUNE"
    )
    
    # Toplam eğitim metriklerini birleştir
    all_train_losses = train_losses_s1 + train_losses_s2
    all_train_accs = train_accs_s1 + train_accs_s2
    all_valid_losses = valid_losses_s1 + valid_losses_s2
    all_valid_accs = valid_accs_s1 + valid_accs_s2
    
    # Grafikleri çiz
    plt.figure(figsize=(15, 10))
    
    # Train Loss
    plt.subplot(2, 3, 1)
    plt.plot(all_train_losses, label='Training Loss', linewidth=2)
    plt.axvline(x=len(train_losses_s1), color='r', linestyle='--', label='Aşama 1/2 Geçiş')
    plt.title('Eğitim Loss Değişimi (Her 2 Aşama)', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    # Valid Loss
    plt.subplot(2, 3, 2)
    plt.plot(all_valid_losses, label='Validation Loss', color='orange', linewidth=2)
    plt.axvline(x=len(valid_losses_s1), color='r', linestyle='--', label='Aşama 1/2 Geçiş')
    plt.title('Validasyon Loss Değişimi (Her 2 Aşama)', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    # Train Accuracy
    plt.subplot(2, 3, 3)
    plt.plot(all_train_accs, label='Training Accuracy', color='green', linewidth=2)
    plt.axvline(x=len(train_accs_s1), color='r', linestyle='--', label='Aşama 1/2 Geçiş')
    plt.title('Eğitim Accuracy Değişimi (Her 2 Aşama)', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy (%)')
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    # Valid Accuracy
    plt.subplot(2, 3, 4)
    plt.plot(all_valid_accs, label='Validation Accuracy', color='red', linewidth=2)
    plt.axvline(x=len(valid_accs_s1), color='r', linestyle='--', label='Aşama 1/2 Geçiş')
    plt.title('Validasyon Accuracy Değişimi (Her 2 Aşama)', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy (%)')
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    # Aşama 1 Karşılaştırması
    plt.subplot(2, 3, 5)
    plt.plot(train_accs_s1, label='Train', marker='o')
    plt.plot(valid_accs_s1, label='Valid', marker='s')
    plt.title('Aşama 1: Head Training', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy (%)')
    plt.legend()
    plt.grid(True, alpha=0.3)

    plt.plot(train_accs_s2, label='Train', marker='o')
    plt.plot(valid_accs_s2, label='Valid', marker='s')
    plt.title('Aşama 2: Fine-Tuning', fontsize=12, fontweight='bold')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy (%)')
    plt.legend()

    print(f"Aşama 2 Final Accuracy: {train_accs_s2[-1]:.2f}% (Train), {valid_accs_s2[-1]:.2f}% (Valid)")
